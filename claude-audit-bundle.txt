  maxPrice: number;
}

export default function ConfiguratorPage() {
  // Input dimensions & specs
  const [system, setSystem] = useState<SystemType>("upvc");
  const [widthInput, setWidthInput] = useState<string>("180");
  const [heightInput, setHeightInput] = useState<string>("160");
  const [panes, setPanes] = useState<number>(2);
  const [hasTransom, setHasTransom] = useState<boolean>(false);
  const [transomHeightInput, setTransomHeightInput] = useState<string>("45");
  const [openingType, setOpeningType] = useState<OpeningType>("tilt-turn");
  const [frameColor, setFrameColor] = useState<FrameColor>("white");
  const [glassType, setGlassType] = useState<string>("double-lowe");
  const [installation, setInstallation] = useState<string>("renovation");

  // Cart & Project metadata
  const [cart, setCart] = useState<CartItem[]>(() => {
    if (typeof window === "undefined") return [];
    try {
      const saved = localStorage.getItem("sepanta_quote_cart");
      return saved ? JSON.parse(saved) : [];
    } catch {
      return [];
    }
  });

  const [planFileName, setPlanFileName] = useState<string>("elevation-facade.pdf");
  const [clientName, setClientName] = useState<string>("");
  const [clientPhone, setClientPhone] = useState<string>("");
  const [isAddedToCartFeedback, setIsAddedToCartFeedback] = useState<boolean>(false);
  const quoteMeta = {
    id: "۴۸۲۹۰۱",
    date: "۱۴۰۳/۰۹/۰۳",
  };

  // Save cart to localStorage
  const saveCart = (items: CartItem[]) => {
    setCart(items);
    try {
      localStorage.setItem("sepanta_quote_cart", JSON.stringify(items));
    } catch {
      // Ignore storage errors
    }
  };

  // Numeric sanitization
  const widthCm = Math.max(40, Math.min(600, parseInt(toEnglishDigits(widthInput) || "180", 10)));
  const heightCm = Math.max(40, Math.min(400, parseInt(toEnglishDigits(heightInput) || "160", 10)));
  const transomHeightCm = Math.max(25, Math.min(90, parseInt(toEnglishDigits(transomHeightInput) || "45", 10)));

  // Engineering calculation logic
  const areaSqm = (widthCm * heightCm) / 10000;
  const sashWidth = widthCm / panes;

  // System base rate per sqm
  const SYSTEM_RATES: Record<SystemType, number> = {
    upvc: 3900000,
    "thermal-break": 8500000,
    "balcony-glass": 4500000,
    "pleated-mesh": 1450000,
  };

  const COLOR_MULTIPLIERS: Record<FrameColor, number> = {
    white: 1.0,
    anthracite: 1.12,
    champagne: 1.18,
    wood: 1.22,
  };

  const GLASS_ADDONS: Record<string, number> = {
    "double-clear": 0,
    "double-lowe": 450000,
    "double-laminated": 950000,
    triple: 1400000,
  };

  const HARDWARE_ADDONS: Record<OpeningType, number> = {
    fixed: 0,
    turn: 480000,
    "tilt-turn": 880000,
    sliding: 750000,
    folding: 600000,
  };

  const baseSystemCost = areaSqm * SYSTEM_RATES[system];
  const colorFactor = COLOR_MULTIPLIERS[frameColor];
  const glassCost = system === "pleated-mesh" ? 0 : areaSqm * (GLASS_ADDONS[glassType] || 0);
  const hardwareCost = HARDWARE_ADDONS[openingType] * (panes > 1 ? Math.floor(panes / 2) : 1);
  const transomCost = hasTransom ? 450000 : 0;
  const installMultiplier = installation === "renovation" ? 1.12 : installation === "new" ? 1.08 : 1.0;

  const medianPrice =
    (baseSystemCost * colorFactor + glassCost + hardwareCost + transomCost) * installMultiplier;

  const minPrice = Math.round((medianPrice * 0.92) / 10000) * 10000;
  const maxPrice = Math.round((medianPrice * 1.08) / 10000) * 10000;

  // Engineering Validation Warnings
  const warnings: string[] = [];
  if (sashWidth > 100 && (openingType === "turn" || openingType === "tilt-turn")) {
    warnings.push(
      `عرض هر لنگه (${toPersianDigits(Math.round(sashWidth))} سانتی‌متر) بیش از ۱۰۰ سانتی‌متر است. جهت جلوگیری از افتادگی لولا، پیشنهاد می‌شود تعداد لنگه را افزایش داده یا از سیستم کشویی/ترمال بریک استفاده کنید.`
    );
  }
  if (heightCm > 220 && !hasTransom) {
    warnings.push(
      `ارتفاع دهانه (${toPersianDigits(heightCm)} سانتی‌متر) بیش از ۲۲۰ سانتی‌متر است. پیشنهاد مهندسی سپنتا، افزودن کتیبه بالا (Transom) یا استفاده از مقاطع مقاوم آلومینیوم ترمال بریک است.`
    );
  }
  if (areaSqm > 3.2) {
    warnings.push(
      `مساحت کل دهانه (${toPersianDigits(areaSqm.toFixed(2))} مترمربع) از ۳.۲ مترمربع بیشتر است. طبق مبحث ۴ مقررات ملی، استفاده از شیشه سکوریت یا لمینت ایمنی پیشنهاد اکید می‌شود.`
    );

    "pleated-mesh": "توری کشویی پلیسه",
  };

      id: "item-" + Date.now(),
      name: `دهانه ${toPersianDigits(widthCm)}×${toPersianDigits(heightCm)} (${systemTitles[system]})`,
      system,
      systemTitle: systemTitles[system],
      widthCm,
      heightCm,
      panes,
      hasTransom,
      openingType,
      frameColor,
      glassType,
      installation,
      minPrice,
      maxPrice,
    };
    saveCart([...cart, newItem]);
    setIsAddedToCartFeedback(true);
    setTimeout(() => setIsAddedToCartFeedback(false), 2500);
  };

    saveCart(cart.filter((c) => c.id !== id));
  };

  const handleClearCart = () => {
    saveCart([]);
  };

  const totalMin = cart.reduce((acc, item) => acc + item.minPrice, 0);
  const totalMax = cart.reduce((acc, item) => acc + item.maxPrice, 0);

import React from "react";
import { toPersianDigits } from "@/lib/format";

export type SystemType = "upvc" | "thermal-break" | "balcony-glass" | "pleated-mesh";
export type OpeningType = "fixed" | "turn" | "tilt-turn" | "sliding" | "folding";
export type FrameColor = "white" | "anthracite" | "champagne" | "wood";

export interface WindowStudioProps {
  widthCm: number;
  heightCm: number;
  panes: number; // 1 to 4
  hasTransom: boolean;
  transomHeightCm: number;
  system: SystemType;
  openingType: OpeningType;
  frameColor: FrameColor;
}

const COLOR_MAP: Record<FrameColor, { fill: string; stroke: string; label: string }> = {
  white: { fill: "#F8FAFC", stroke: "#CBD5E1", label: "سفید استاندارد" },
  anthracite: { fill: "#1E242B", stroke: "#0F1216", label: "آنتراسیت مات" },
  champagne: { fill: "#B8A58D", stroke: "#8E7B65", label: "شامپاینی متالیک" },
  wood: { fill: "#6A3F1F", stroke: "#452610", label: "طرح چوب بلوطی" },
};

export function WindowStudioSVG({
  widthCm,
  heightCm,
  panes,
  hasTransom,
  transomHeightCm,
  system,
  openingType,
  frameColor,
}: WindowStudioProps) {
  // Normalize SVG canvas coordinates with padding for architectural dimension lines
  const paddingX = 70;
  const paddingY = 70;
  const maxViewWidth = 800;
  const maxViewHeight = 550;

  const innerAreaWidth = maxViewWidth - paddingX * 2;
  const innerAreaHeight = maxViewHeight - paddingY * 2;

  // Aspect ratio calculation
  const ratio = widthCm / heightCm;
  let frameWidth = innerAreaWidth;
  let frameHeight = frameWidth / ratio;

  if (frameHeight > innerAreaHeight) {
    frameHeight = innerAreaHeight;
    frameWidth = frameHeight * ratio;
  }

  const frameX = paddingX + (innerAreaWidth - frameWidth) / 2;
  const frameY = paddingY + (innerAreaHeight - frameHeight) / 2;

  const colorData = COLOR_MAP[frameColor] || COLOR_MAP.white;
  const isDarkFrame = frameColor === "anthracite" || frameColor === "wood";

  // Transom calculation
  const transomRatio = hasTransom ? Math.min(0.4, Math.max(0.15, transomHeightCm / heightCm)) : 0;
  const transomHeight = frameHeight * transomRatio;
  const mainHeight = frameHeight - transomHeight;
  const mainY = frameY + transomHeight;

  // Pane width
  const paneWidth = frameWidth / panes;
  const outerBorderWidth = system === "thermal-break" ? 14 : system === "balcony-glass" ? 10 : 16;
  const sashBorderWidth = 10;

  return (
    <div className="w-full flex flex-col items-center justify-center select-none">
      <svg
        viewBox={`0 0 ${maxViewWidth} ${maxViewHeight}`}
        className="w-full h-auto max-h-[480px] drop-shadow-xl"
        aria-label="پیش‌نمایش معماری دو بعدی دهانه پنجره"
      >
        <defs>
          {/* Subtle realistic glass sheen gradient */}
          <linearGradient id="glassGradient" x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" stopColor="#e0f2fe" stopOpacity="0.45" />
            <stop offset="40%" stopColor="#bae6fd" stopOpacity="0.18" />
            <stop offset="42%" stopColor="#ffffff" stopOpacity="0.55" />
            <stop offset="48%" stopColor="#bae6fd" stopOpacity="0.2" />
            <stop offset="100%" stopColor="#7dd3fc" stopOpacity="0.35" />
          </linearGradient>

          {/* Pleated mesh texture pattern */}
          <pattern id="meshGrid" width="6" height="6" patternUnits="userSpaceOnUse">
            <path d="M 0 3 L 3 0 L 6 3 L 3 6 Z" fill="none" stroke="#475569" strokeWidth="0.75" />
          </pattern>
        </defs>

        {/* Outer Frame Wall Niche Background */}
        <rect
          x={frameX - 6}
          y={frameY - 6}
          width={frameWidth + 12}
          height={frameHeight + 12}
          fill="#0B0E11"
          opacity="0.08"
        />

        {/* Main Outer Profile Frame */}
        <rect
          x={frameX}
          y={frameY}
          width={frameWidth}
          height={frameHeight}
          fill={colorData.fill}
          stroke={colorData.stroke}
          strokeWidth="2"
        />

        {/* Transom (کتیبه بالا) if enabled */}
        {hasTransom && (
          <g id="transom-section">
            <rect
              x={frameX + outerBorderWidth}
              y={frameY + outerBorderWidth}
              width={frameWidth - outerBorderWidth * 2}
              height={transomHeight - outerBorderWidth}
              fill="url(#glassGradient)"
              stroke={colorData.stroke}
              strokeWidth="1.5"
            />
            {/* Horizontal transom divider bar */}
            <rect
              x={frameX}
              y={frameY + transomHeight - 4}
              width={frameWidth}
              height={outerBorderWidth}
              fill={colorData.fill}
              stroke={colorData.stroke}
              strokeWidth="1.5"
            />
            {/* Glass diagonal glare line */}
            <line
              x1={frameX + outerBorderWidth + 10}
              y1={frameY + outerBorderWidth + 5}
              x2={frameX + frameWidth - outerBorderWidth - 10}
              y2={frameY + transomHeight - 10}
              stroke="#ffffff"
              strokeWidth="2"
              strokeOpacity="0.6"
            />
          </g>
        )}

        {/* Sashes & Glass Panes */}
        {Array.from({ length: panes }).map((_, i) => {
          const pX = frameX + i * paneWidth;
          const pY = mainY;
          const pW = paneWidth;
          const pH = mainHeight;

          // In standard multi-sash windows, alternate fixed and operable
          const isOperable =
            system === "balcony-glass" ||
            system === "pleated-mesh" ||
            (panes === 1 ? openingType !== "fixed" : i % 2 === 1 ? openingType !== "fixed" : false);

          const innerGlassX = pX + outerBorderWidth;
          const innerGlassY = pY + (hasTransom ? outerBorderWidth : outerBorderWidth);
          const innerGlassW = pW - outerBorderWidth * 2;
          const innerGlassH = pH - outerBorderWidth * 2;

          return (
            <g key={i} id={`pane-${i}`}>
              {/* Vertical Mullion Divider between panes */}
              {i > 0 && (
                <rect
                  x={pX - outerBorderWidth / 2}
                  y={mainY}
                  width={outerBorderWidth}
                  height={pH}
                  fill={colorData.fill}
                  stroke={colorData.stroke}
                  strokeWidth="1.5"
                />
              )}

              {/* Sash Profile if operable */}
              {isOperable && system !== "balcony-glass" && system !== "pleated-mesh" && (
                <rect
                  x={innerGlassX}
                  y={innerGlassY}
                  width={innerGlassW}
                  height={innerGlassH}
                  fill={colorData.fill}
                  stroke={colorData.stroke}
                  strokeWidth="1"
                />
              )}

              {/* Glass Area or Mesh */}
              <rect
                x={innerGlassX + (isOperable ? sashBorderWidth : 0)}
                y={innerGlassY + (isOperable ? sashBorderWidth : 0)}
                width={innerGlassW - (isOperable ? sashBorderWidth * 2 : 0)}
                height={innerGlassH - (isOperable ? sashBorderWidth * 2 : 0)}
                fill={system === "pleated-mesh" ? "url(#meshGrid)" : "url(#glassGradient)"}
                stroke="#94A3B8"
                strokeWidth="0.75"
              />

              {/* Diagonal Glass Sheen Reflection */}
              {system !== "pleated-mesh" && (
                <line
                  x1={innerGlassX + 15}
                  y1={innerGlassY + 15}
                  x2={innerGlassX + innerGlassW - 15}
                  y2={innerGlassY + innerGlassH - 15}
                  stroke="#ffffff"
                  strokeWidth="2.5"
                  strokeOpacity="0.5"
                />
              )}

              {/* Architectural Opening Direction Symbols */}
              {isOperable && (
                <g id={`opening-symbol-${i}`}>
                  {/* Single Turn (Side Hung): Triangle pointing to handle side */}
                  {openingType === "turn" && (
                    <path
                      d={`M ${innerGlassX + 10} ${innerGlassY + 10} L ${innerGlassX + innerGlassW - 10} ${innerGlassY + innerGlassH / 2} L ${innerGlassX + 10} ${innerGlassY + innerGlassH - 10}`}
                      fill="none"
                      stroke="#C27803"
                      strokeWidth="2"
                      strokeDasharray="4 3"
                    />
                  )}

                  {/* Tilt & Turn (Dual action): Solid bottom tilt triangle + side dashed triangle */}
                  {openingType === "tilt-turn" && (
                    <>
                      <path
                        d={`M ${innerGlassX + 10} ${innerGlassY + 10} L ${innerGlassX + innerGlassW - 10} ${innerGlassY + innerGlassH / 2} L ${innerGlassX + 10} ${innerGlassY + innerGlassH - 10}`}
                        fill="none"
                        stroke="#C27803"
                        strokeWidth="1.75"
                        strokeDasharray="4 3"
                      />
                      <path
                        d={`M ${innerGlassX + 10} ${innerGlassY + innerGlassH - 10} L ${innerGlassX + innerGlassW / 2} ${innerGlassY + 10} L ${innerGlassX + innerGlassW - 10} ${innerGlassY + innerGlassH - 10}`}
                        fill="none"
                        stroke="#0D9488"
                        strokeWidth="1.75"
                        strokeDasharray="3 3"
                      />
                    </>
                  )}

                  {/* Sliding (Horizontal arrow) */}
                  {openingType === "sliding" && (
                    <g transform={`translate(${innerGlassX + innerGlassW / 2 - 20}, ${innerGlassY + innerGlassH / 2 - 8})`}>
                      <rect width="40" height="16" fill="#0B0E11" rx="2" opacity="0.8" />
                      <path d="M 6 8 L 34 8 M 12 4 L 6 8 L 12 12 M 28 4 L 34 8 L 28 12" stroke="#FFFFFF" strokeWidth="1.5" fill="none" />
                    </g>
                  )}

                  {openingType === "folding" && (
                    <g transform={`translate(${innerGlassX + innerGlassW / 2 - 25}, ${innerGlassY + innerGlassH / 2 - 10})`}>
                      <rect width="50" height="20" fill="#0B0E11" rx="2" opacity="0.8" />
                      <path d="M 8 16 L 16 4 L 24 16 L 32 4 L 40 16" stroke="#C27803" strokeWidth="2" fill="none" />

  const [sliderPos, setSliderPos] = useState<number>(50); // percentage 0 - 100
  const containerRef = useRef<HTMLDivElement>(null);
  const isDragging = useRef<boolean>(false);

  const handleMove = useCallback((clientX: number) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const x = clientX - rect.left;
    const percentage = Math.max(0, Math.min(100, (x / rect.width) * 100));
    setSliderPos(percentage);
  }, []);

  const onTouchMove = (e: React.TouchEvent) => {
    if (e.touches[0]) {
      handleMove(e.touches[0].clientX);
    }
  };

  const onMouseMove = (e: React.MouseEvent) => {
    if (isDragging.current) {
      handleMove(e.clientX);
    }
  };

/**
 * Returns a product by its slug or id.
 */
export function getProductBySlug(slug: string): ProductItem | undefined {
  return PRODUCTS.find((p) => p.slug === slug || p.id === slug);
}

/**
 * Returns all products belonging to a given cluster.
 */
export function getProductsByCluster(cluster: ProductClusterId): ProductItem[] {
  return PRODUCTS.filter((p) => p.cluster === cluster);
}

/**
 * Returns all products that support the live interactive configurator.
 */
export function getConfigurableProducts(): ProductItem[] {
  return PRODUCTS.filter((p) => p.configuratorAvailable);
}

const isProd = process.env.NODE_ENV === "production";
export const BASE_PATH = isProd ? "/sepanta-co" : "";

/**
 * Returns a static asset path prefixed with GitHub Pages basePath in production.
 * @example assetPath("/images/hero.webp") -> "/sepanta-co/images/hero.webp" (in prod)
 */
export function assetPath(path: string): string {
  if (!path) return "";
  // If already absolute URL or data URI, return as-is
  if (path.startsWith("http://") || path.startsWith("https://") || path.startsWith("data:")) {
    return path;
  }
  const cleanPath = path.startsWith("/") ? path : `/${path}`;
  return `${BASE_PATH}${cleanPath}`;
}

export const getAssetPath = assetPath;

/**
 * Persian typography & formatting utilities for Sepanta Industrial Group
 */

const PERSIAN_DIGITS = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];
const ARABIC_DIGITS = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"];

/**
 * Converts English digits (0-9) to standard Persian digits (۰-۹).
 */
export function toPersianDigits(input: string | number): string {
  if (input === null || input === undefined) return "";
  const str = String(input);
  return str.replace(/\d/g, (d) => PERSIAN_DIGITS[parseInt(d, 10)]);
}

/**
 * Normalizes both Persian (۰-۹) and Arabic (٠-٩) digits to standard ASCII English digits (0-9).
 * Crucial for technical form inputs, calculator dimensions, and math calculations.
 */
export function toEnglishDigits(input: string): string {
  if (!input) return "";
  let result = input;
  for (let i = 0; i < 10; i++) {
    result = result.replace(new RegExp(PERSIAN_DIGITS[i], "g"), String(i));
    result = result.replace(new RegExp(ARABIC_DIGITS[i], "g"), String(i));
  }
  return result;
}

/**
 * Normalizes Arabic characters (ي / ك) to standard Persian (ی / ک)
 * and cleans up common non-standard spacing while preserving ZWNJ.
 */
export function normalizePersianText(text: string): string {
  if (!text) return "";
  return text
    .replace(/ي/g, "ی")
    .replace(/ك/g, "ک")
    .replace(/ۀ/g, "ه‌ی");
}

/**
 * Formats a number with standard Persian thousand separators (٬).
 */
export function formatPersianNumber(num: number): string {
  if (isNaN(num)) return "۰";
  const parts = Math.round(num).toString().split(".");
  parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, "٬");
  return toPersianDigits(parts.join("."));
}

/**
 * Formats a price range in Toman with Persian digits.
 * @example formatTomanRange(3500000, 4800000) => "۳٬۵۰۰٬۰۰۰ تا ۴٬۸۰۰٬۰۰۰ تومان"
 */
export function formatTomanRange(min: number, max: number): string {
  const minStr = formatPersianNumber(min);
  const maxStr = formatPersianNumber(max);
  if (min === max) {
    return `${minStr} تومان`;
  }
  return `${minStr} تا ${maxStr} تومان`;
}
